def evaluate_llm_documentation():
    """
    Framework for evaluating documentation generated by LLMs
    """
    evaluation_criteria = {
        # Accuracy metrics
        'factual_correctness': {
            'description': 'Check if technical information and facts are accurate',
            'score': 0
        },
        'code_accuracy': {
            'description': 'Verify if code examples work as described',
            'score': 0
        },

        # Completeness metrics
        'coverage': {
            'description': 'Check if all key aspects are documented',
            'score': 0
        },
        'examples': {
            'description': 'Assess quality and relevance of examples',
            'score': 0
        },

        # Technical metrics
        'consistency': {
            'description': 'Check terminology and style consistency',
            'score': 0
        },
        'api_alignment': {
            'description': 'Verify alignment with actual API/code',
            'score': 0
        }
    }

    def score_criterion(criterion, score):
        """Score a criterion from 0-10"""
        if 0 <= score <= 10:
            evaluation_criteria[criterion]['score'] = score
        else:
            raise ValueError("Score must be between 0 and 10")

    def calculate_overall_score():
        """Calculate average score across all criteria"""
        scores = [c['score'] for c in evaluation_criteria.values()]
        return sum(scores) / len(scores)

    def generate_report():
        """Generate evaluation report"""
        report = []
        report.append("Documentation Evaluation Report")
        report.append("===========================")

        for criterion, details in evaluation_criteria.items():
            report.append(f"\n{criterion}:")
            report.append(f"Description: {details['description']}")
            report.append(f"Score: {details['score']}/10")

        report.append(f"\nOverall Score: {calculate_overall_score():.2f}/10")
        return "\n".join(report)

    return {
        'criteria': evaluation_criteria,
        'score_criterion': score_criterion,
        'calculate_overall_score': calculate_overall_score,
        'generate_report': generate_report
    }
